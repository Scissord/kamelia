# Создать миграцию
migrate create -ext sql -dir internal/migrations user

# Применить миграции
migrate -path "internal/migrations" -database "postgres://postgres:322434@localhost:5432/kamelia?sslmode=disable" up

# Откатить последнюю миграцию
migrate -path "internal/migrations" -database "postgres://postgres:322434@localhost:5432/kamelia?sslmode=disable" down
